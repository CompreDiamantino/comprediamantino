rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // LOJISTAS — leitura e cadastro públicos; update permitido (site ainda sem Firebase Auth);
    // delete bloqueado para evitar apagar tudo
    match /lojistas/{id} {
      allow read: if true;
      allow create: if true;
      allow update: if true;
      allow delete: if false;
    }

    match /posts/{id} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /avaliacoes/{id} {
      allow read: if true;
      allow create: if true;
      allow update, delete: if request.auth != null;
    }

    match /_sistema/{doc} {
      allow read, write: if true;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
